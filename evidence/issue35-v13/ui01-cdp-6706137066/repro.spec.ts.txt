import { defaultDisplayPreferences } from '../src/canonical/displayPreferences'
import { readFileSync } from 'node:fs'
import { readCanonicalState } from './helpers/state'
import { canonicalFixture } from '../src/canonical/fixtures'
import { expect, test, type Page } from '@playwright/test'

test.use({ viewport: { width: 360, height: 800 } })

async function seed(page: Page, fixture = canonicalFixture) {
  await page.clock.setFixedTime(new Date('2026-09-24T12:00:00+09:00'))
  await page.goto('/')
  await page.evaluate(async data => {
    const db = await new Promise<IDBDatabase>((resolve, reject) => {
      const request = indexedDB.open('training-check', 7)
      request.onsuccess = () => resolve(request.result)
      request.onerror = () => reject(request.error)
    })
    await new Promise<void>((resolve, reject) => {
      const tx = db.transaction(['canonical', 'cutover'], 'readwrite')
      tx.objectStore('canonical').put(data, 'app')
      tx.objectStore('cutover').put({ authoritative: true, establishedAt: '2026-09-24T00:00:00+09:00' }, 'state')
      tx.oncomplete = () => resolve()
      tx.onabort = () => reject(tx.error)
    })
    db.close()
  }, fixture)
  await page.reload()
  await expect(page.getByRole('heading', { name: '今週の実施メニュー' })).toBeVisible()
}


test('UI-01 C1-C3 CDP gesture contrasts', async ({ browser, baseURL }, info) => {
  const results: any[] = []
  const execute = async (name: string, kind: 'static' | 'move' | 'handle') => {
    const drag = kind === 'handle'
    const context = await browser.newContext({ baseURL, viewport: { width: 360, height: 400 }, hasTouch: true, isMobile: true, timezoneId: 'Asia/Tokyo' })
    const row: any = { name, kind, messages: [] }
    results.push(row)
    try {
      const page = await context.newPage()
      page.on('pageerror', error => row.messages.push({ type: 'pageerror', text: error.message }))
      page.on('console', message => row.messages.push({ type: message.type(), text: message.text() }))
      const fixture = {
        ...structuredClone(canonicalFixture),
        trainingItems: canonicalFixture.trainingItems.map(item => ({ ...item, managementMode: 'MAN' as const })),
        menus: canonicalFixture.menus.map(menu => ({ ...menu, managementMode: 'MAN' as const })),
        displayPreferences: structuredClone(defaultDisplayPreferences), appliedProposalIds: [],
      }
      await seed(page, fixture)
      await page.getByRole('button', { name: '共通メニュー' }).tap()
      await page.getByRole('button', { name: '設定', exact: true }).tap()
      await page.getByRole('button', { name: /^表示設定/ }).tap()
      const group = page.getByRole('region', { name: 'Default', exact: true })
      const option = page.getByRole('region', { name: 'Option', exact: true })
      const handles = group.locator('.sort-drag-handle')
      await expect(handles).toHaveCount(3)
      const optionBefore = await option.textContent()
      row.before = await readCanonicalState(page)
      await page.evaluate(() => {
        const events: object[] = [], idb: object[] = []
        Object.assign(window, { contrastProbe: { events, idb } })
        const types = ['pointerdown','pointermove','pointerup','pointercancel','gotpointercapture','lostpointercapture','touchstart','touchmove','touchend','touchcancel','mousedown','mouseup','click']
        const tag = (node: EventTarget | null) => node instanceof Element ? { tag: node.tagName, text: node.textContent?.trim().slice(0, 70) } : null
        for (const type of types) for (const capture of [true, false]) document.addEventListener(type, event => {
          const pointer = event as PointerEvent & { changedTouches?: TouchList }
          const touch = pointer.changedTouches?.[0]
          const x = touch?.clientX ?? pointer.clientX, y = touch?.clientY ?? pointer.clientY
          const target = event.target as Element
          const record = {
            type, capture, time: event.timeStamp, trusted: event.isTrusted, pointerId: pointer.pointerId,
            target: tag(target), currentTarget: 'document', hit: tag(document.elementFromPoint(x, y)), x, y,
            prevented: event.defaultPrevented, preventedAfter: false, scroll: window.scrollY,
            touchAction: target instanceof Element ? getComputedStyle(target).touchAction : null,
            pointerCaptured: target instanceof Element && Number.isInteger(pointer.pointerId) ? target.hasPointerCapture(pointer.pointerId) : null,
          }
          events.push(record)
          queueMicrotask(() => { record.preventedAfter = event.defaultPrevented })
        }, { capture, passive: true })
        const get = IDBObjectStore.prototype.get, put = IDBObjectStore.prototype.put
        IDBObjectStore.prototype.get = function (...args) {
          idb.push({ action: 'get', store: this.name, mode: this.transaction.mode })
          return get.apply(this, args)
        }
        IDBObjectStore.prototype.put = function (...args) {
          idb.push({ action: 'put', store: this.name, mode: this.transaction.mode })
          for (const type of ['complete', 'abort']) this.transaction.addEventListener(type, () => idb.push({ action: type }), { once: true })
          return put.apply(this, args)
        }
      })
      const snapshot = () => page.evaluate(() => ({
        heading: document.querySelector('h1')?.textContent,
        alerts: [...document.querySelectorAll('[role=status],[role=alert]')].map(node => node.textContent),
        scroll: window.scrollY, probe: Reflect.get(window, 'contrastProbe'),
      }))
      const tap = async (button: ReturnType<Page['getByRole']>) => {
        await button.scrollIntoViewIfNeeded()
        row.target = await button.evaluate(node => {
          const rect = node.getBoundingClientRect()
          const hit = document.elementFromPoint(rect.x + rect.width / 2, rect.y + rect.height / 2)
          return { rect: rect.toJSON(), hit: hit?.textContent?.trim(), match: node.contains(hit), disabled: (node as HTMLButtonElement).disabled, scroll: window.scrollY }
        })
        expect(row.target.match).toBe(true)
        expect(row.target.disabled).toBe(false)
        await button.tap()
      }
      await handles.last().scrollIntoViewIfNeeded()
      const cdp = await context.newCDPSession(page)
      const center = async (index: number) => {
        const rect = (await handles.nth(index).boundingBox())!
        return { x: rect.x + rect.width / 2, y: rect.y + rect.height / 2 }
      }
      const [first, last] = await Promise.all([center(0), center(2)])
      const start = drag ? last : { x: 340, y: last.y }
      const end = drag ? first : { x: 340, y: last.y + first.y - last.y }
      row.gesture = await page.evaluate(({ start, end }) => ({
        scroll: window.scrollY, start, end,
        hits: [start, end].map(({ x, y }) => {
          const hit = document.elementFromPoint(x, y)
          return { tag: hit?.tagName, class: hit?.className, handle: !!hit?.closest('.sort-drag-handle'),
            interactive: !!hit?.closest('button,input,select,a,label'), touchAction: hit ? getComputedStyle(hit).touchAction : null }
        }),
      }), { start, end })
      expect(row.gesture.hits.every((hit: { handle: boolean; interactive: boolean }) => hit.handle === drag && !hit.interactive)).toBe(true)
      await cdp.send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [start] })
      if (kind !== 'static') await cdp.send('Input.dispatchTouchEvent', { type: 'touchMove', touchPoints: [end] })
      await cdp.send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] })
      await expect(group.locator('.sort-priority-row span:first-child')).toHaveText(drag
        ? ['1 部位', '2 推奨曜日', '3 実施状況']
        : ['1 推奨曜日', '2 実施状況', '3 部位'])
      await page.evaluate(() => new Promise<void>(resolve => requestAnimationFrame(() => requestAnimationFrame(() => resolve()))))
      row.gesture.afterScroll = await page.evaluate(() => window.scrollY)
      if (drag) expect(row.gesture.afterScroll).toBe(row.gesture.scroll)
      expect(await option.textContent()).toBe(optionBefore)
      row.draft = await page.locator('.display-preference-section').evaluateAll(sections => sections.map(section => ({
        priorities: [...section.querySelectorAll('.sort-priority-row span:first-child')].map(node => node.textContent),
        direction: section.querySelector('input:checked')?.parentElement?.textContent,
        pending: [...section.querySelectorAll('fieldset')].some(node => node.disabled),
      })))
      await tap(page.getByRole('button', { name: /戻る/ }))
      try { await expect(page.getByRole('heading', { name: '設定', exact: true })).toBeVisible({ timeout: 3000 }) }
      catch { row.routeTimeout = true }
      row.after = await readCanonicalState(page)
      row.state = await snapshot()
      expect(row.after).toEqual(row.before)
      row.click = row.state.probe.events.some((event: { type: string; target?: { text?: string } }) =>
        event.type === 'click' && event.target?.text?.includes('戻る'))
      row.passed = row.click && !row.routeTimeout
    } catch (error) {
      row.passed = false
      row.error = String(error)
      const page = context.pages()[0]
      if (page) {
        row.stopped = await page.evaluate(() => ({ heading: document.querySelector('h1')?.textContent, probe: Reflect.get(window, 'contrastProbe') }))
        row.after = await readCanonicalState(page)
      }
    } finally { await context.close() }
    return row.passed
  }
  await execute('C1 static', 'static')
  await execute('C2 movement off handle', 'move')
  await execute('C3 handle drag', 'handle')
  await info.attach('ui01-cdp-contrast', { body: JSON.stringify(results), contentType: 'application/json' })
  for (const result of results) expect.soft(result.passed, result.name).toBe(true)
})
